cmake_minimum_required(VERSION 3.16)
project(RAView LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 REQUIRED COMPONENTS Core Quick)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GUI REQUIRED IMPORTED_TARGET gz-gui8)
pkg_check_modules(RENDER REQUIRED IMPORTED_TARGET gz-rendering8)
pkg_check_modules(TRANSPORT REQUIRED IMPORTED_TARGET gz-transport13)
add_library(RAView SHARED RAView.cc RAView.qrc)
target_link_libraries(RAView PRIVATE PkgConfig::GUI PkgConfig::RENDER PkgConfig::TRANSPORT Qt5::Core Qt5::Quick)

pkg_check_modules(SIM REQUIRED IMPORTED_TARGET gz-sim8)
add_library(ra_obstacle_motion SHARED ../RaObstacleMotion.cc)
target_link_libraries(ra_obstacle_motion PRIVATE PkgConfig::SIM PkgConfig::TRANSPORT)
