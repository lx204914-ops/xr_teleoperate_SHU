cmake_minimum_required(VERSION 3.16)
project(ra_runtime_dependencies LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SIM REQUIRED IMPORTED_TARGET gz-sim8)
pkg_check_modules(TRANSPORT REQUIRED IMPORTED_TARGET gz-transport13)
add_library(ra_obstacle_motion SHARED code/RaObstacleMotion.cc)
target_link_libraries(ra_obstacle_motion PRIVATE PkgConfig::SIM PkgConfig::TRANSPORT)
add_library(periodic_publish75 SHARED code/periodic_publish75.cc)
